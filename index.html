مسح الحقول بعد التسجيل لتجهيزها لطالب آخر entry_name.delete(0, tk.END) combo_dept.set('') # هنا يمكنك إضافة كود انتقال الطالب لنافذة الأسئلة الخاصة بالاختبار الفعلي def show_student_details(student): # دالة تظهر عند ضغط الأدمن على زر الطالب لمعرفة تفاصيله messagebox.showinfo( "بيانات الطالب", f"الاسم الثلاثي: {student['name']}\nالقسم: {student['department']}\nتقييم النظام: مستعد للاختبار ومجتهد!" ) def open_admin_panel(): # حماية لوحة الإدارة بكلمة مرور لمنع الطلاب من الدخول password = simpledialog.askstring("حماية", "أدخل كلمة مرور المشرف:", show='*') # كلمة المرور الافتراضية هي admin123 if password != "admin123": if password is not None: # لتجاهل الإشعار إذا ضغط المستخدم على "إلغاء" messagebox.showerror("خطأ", "كلمة المرور غير صحيحة، الوصول مرفوض!") return # إنشاء نافذة الإدارة الخاصة بالأدمن admin_window = tk.Toplevel(root) admin_window.title("لوحة الإدارة - بيانات الطلاب") admin_window.geometry("500x600") admin_window.configure(bg="#f0f0f0") tk.Label(admin_window, text="قائمة الطلاب المتقدمين للاختبار", font=("Arial", 16, "bold"), bg="#f0f0f0").pack(pady=20) if not students_database: tk.Label(admin_window, text="لا يوجد طلاب مسجلين في الاختبار حتى الآن.", font=("Arial", 12), bg="#f0f0f0").pack(pady=10) return # توليد زر مستقل لكل طالب مسجل ببياناته for student in students_database: button_text = f"الطالب: {student['name']} | التخصص: {student['department']}" btn = tk.Button( admin_window, text=button_text, font=("Arial", 12, "bold"), bg="#4CAF50", fg="white", cursor="hand2", command=lambda s=student: show_student_details(s) ) btn.pack(fill=tk.X, padx=20, pady=5, ipady=8) # زر الاختبار النهائي للطلاب btn_exam = tk.Button(student_frame, text="تسجيل وبدء الاختبار النهائي", font=("Arial", 14, "bold"), bg="#008CBA", fg="white", cursor="hand2", command=start_exam) btn_exam.pack(fill=tk.X, ipady=8) #
